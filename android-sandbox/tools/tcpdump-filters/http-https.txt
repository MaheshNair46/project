# HTTP/HTTPS Traffic Filter
# Captures HTTP and HTTPS traffic for analysis

# HTTP Traffic (port 80)
tcp port 80

# HTTPS Traffic (port 443)
tcp port 443

# Alternative HTTP/HTTPS ports
tcp port 8080
tcp port 8443
tcp port 3000
tcp port 8000
tcp port 5000

# HTTP/HTTPS over non-standard ports
tcp port 3128  # Proxy
tcp port 1080  # SOCKS proxy
tcp port 8888  # Common alternative HTTP port

# Exclude local traffic
not (src net 192.168.0.0/16 and dst net 192.168.0.0/16)
not (src net 10.0.0.0/8 and dst net 10.0.0.0/8)
not (src net 172.16.0.0/12 and dst net 172.16.0.0/12)

# Focus on outbound connections
src net 192.168.0.0/16 or src net 10.0.0.0/8 or src net 172.16.0.0/12

# Additional notes:
# - This filter captures both HTTP and HTTPS traffic
# - HTTPS traffic will be encrypted, but you can still see metadata
# - Use with -s0 to capture full packets
# - Consider using -A for ASCII output to see HTTP headers
# - For HTTPS analysis, consider using SSL/TLS inspection tools
